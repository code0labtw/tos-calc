const targetLv = parseInt(document.getElementById('targetLv').value) || curLv; const targetRuns = parseInt(document.getElementById('targetRuns').value) || 0; const curTotalExp = getTotalExp(curLv) + curExpInLv; const curStamCap = getStamCap(curLv); const curSpace = getTeamSpace(curLv); document.getElementById('curCapDisplay').textContent = `${curStamCap} / ${curSpace}`; // 模式 A: 設定目標等級 const alertA = document.getElementById('alertA'); if (targetLv <= curLv) { alertA.style.display = 'block'; } else { alertA.style.display = 'none'; } const targetTotalExpA = getTotalExp(targetLv); const expGapA = Math.max(0, targetTotalExpA - curTotalExp); const runsA = Math.ceil(expGapA / expPerRun); const stamA = runsA * stamPerRun; const targetStamCapA = getStamCap(targetLv); const lvDiffA = targetLv - curLv; const refillA = Math.max(0, (stamA - (lvDiffA * curStamCap)) / curStamCap); document.getElementById('resA_expGap').textContent = expGapA.toLocaleString(); document.getElementById('resA_runs').textContent = runsA.toLocaleString() + ' 場'; document.getElementById('resA_stam').textContent = stamA.toLocaleString();
